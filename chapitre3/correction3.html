Correction

from math import *
a=float(input(" taper le nombre a: "))
b=float(input(" taper le nombre b: "))
c=float(input(" taper le nombre c: "))
delta = b*b- 4*a*c
if a==0:
print(" Ce n'est pas une équation du second degré " )
else:
if delta > 0:
x1 = (- b-sqrt(delta))/(2*a)
x2 = (- b+sqrt(delta))/(2*a)
print( f "deux solutions : {x1} et {x2} ")

x = -b/(2*a)
print(f " une seule solution : {x} ")
else:
print(" pas de solution dans R " )
